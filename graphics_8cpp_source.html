185bool UserToNdc(double x, double y, double& x_ndc, double& y_ndc) {
186 if ((gPad->GetLogx() != 0 && x <= 0.0) || (gPad->GetLogy() != 0 && y <= 0.0)) {
187 return false;
188 }
189 if (!std::isfinite(x) || !std::isfinite(y)) {
190 return false;
191 }
192 x_ndc = (gPad->XtoPad(x) - gPad->GetX1()) / (gPad->GetX2() - gPad->GetX1());
193 y_ndc = (gPad->YtoPad(y) - gPad->GetY1()) / (gPad->GetY2() - gPad->GetY1());
194 return true;
195}

209// Occupancy of the frame of the active pad, rasterized in cells of about two pixels.
210class OccupancyGrid {
211 public:
212 OccupancyGrid() {
213 x1_ = gPad->GetLeftMargin();
214 x2_ = 1.0 - gPad->GetRightMargin();
215 y1_ = gPad->GetBottomMargin();
216 y2_ = 1.0 - gPad->GetTopMargin();
217 nx_ = std::clamp(static_cast<int>((x2_ - x1_) * PadWidthPixel() / 2.0), 20, 600);
218 ny_ = std::clamp(static_cast<int>((y2_ - y1_) * PadHeightPixel() / 2.0), 20, 600);
219 cells_.assign(static_cast<size_t>(nx_) * ny_, 0);
220 }
221
222 int nx() const { return nx_; }
223 int ny() const { return ny_; }
224 double CellWidth() const { return (x2_ - x1_) / nx_; }
225 double CellHeight() const { return (y2_ - y1_) / ny_; }
226 double x1() const { return x1_; }
227 double y1() const { return y1_; }
228 double x2() const { return x2_; }
229 double y2() const { return y2_; }
230
231 bool At(int ix, int iy) const { return cells_[static_cast<size_t>(iy) * nx_ + ix] != 0; }
232
233 void MarkRect(double xa, double ya, double xb, double yb) {
234 if (xa > xb) std::swap(xa, xb);
235 if (ya > yb) std::swap(ya, yb);
236 const int ixa = std::max(0, static_cast<int>(std::floor((xa - x1_) / CellWidth())));
237 const int ixb = std::min(nx_ - 1, static_cast<int>(std::floor((xb - x1_) / CellWidth())));
238 const int iya = std::max(0, static_cast<int>(std::floor((ya - y1_) / CellHeight())));
239 const int iyb = std::min(ny_ - 1, static_cast<int>(std::floor((yb - y1_) / CellHeight())));
240 for (int iy = iya; iy <= iyb; ++iy) {
241 for (int ix = ixa; ix <= ixb; ++ix) {
242 cells_[static_cast<size_t>(iy) * nx_ + ix] = 1;
243 }
244 }
245 }
246
247 void MarkPoint(double x, double y, double rx = 0.0, double ry = 0.0) {
248 MarkRect(x - rx, y - ry, x + rx, y + ry);
249 }
250
251 void MarkSegment(double xa, double ya, double xb, double yb) {
252 const double steps =
253 std::max(std::abs(xb - xa) / CellWidth(), std::abs(yb - ya) / CellHeight());
254 if (!std::isfinite(steps)) return;
255 // Long segments far outside the frame are still sampled finely enough inside it.
256 const int n = std::min(200000, static_cast<int>(std::ceil(2.0 * steps)) + 1);
257 for (int i = 0; i <= n; ++i) {
258 const double t = static_cast<double>(i) / n;
259 MarkPoint(xa + t * (xb - xa), ya + t * (yb - ya));
260 }
261 }
262
263 void MarkUserRect(double xa, double ya, double xb, double yb) {
264 double nxa = 0.0, nya = 0.0, nxb = 0.0, nyb = 0.0;
265 if (UserToNdc(xa, ya, nxa, nya) && UserToNdc(xb, yb, nxb, nyb)) {
266 MarkRect(nxa, nya, nxb, nyb);
267 }
268 }
269
270 void MarkUserSegment(double xa, double ya, double xb, double yb) {
271 double nxa = 0.0, nya = 0.0, nxb = 0.0, nyb = 0.0;
272 if (UserToNdc(xa, ya, nxa, nya) && UserToNdc(xb, yb, nxb, nyb)) {
273 MarkSegment(nxa, nya, nxb, nyb);
274 }
275 }

277 // Grow occupied cells by (mx, my) cells in each direction.
278 void Dilate(int mx, int my) {
279 std::vector<char> tmp(cells_.size(), 0);
280 for (int iy = 0; iy < ny_; ++iy) {
281 int last = -1000000;
282 for (int ix = 0; ix < nx_; ++ix) {
283 if (At(ix, iy)) last = ix;
284 if (ix - last <= mx) tmp[static_cast<size_t>(iy) * nx_ + ix] = 1;
285 }
286 last = 1000000;
287 for (int ix = nx_ - 1; ix >= 0; --ix) {
288 if (At(ix, iy)) last = ix;
289 if (last - ix <= mx) tmp[static_cast<size_t>(iy) * nx_ + ix] = 1;
290 }
291 }
292 cells_.assign(cells_.size(), 0);
293 for (int ix = 0; ix < nx_; ++ix) {
294 int last = -1000000;
295 for (int iy = 0; iy < ny_; ++iy) {
296 if (tmp[static_cast<size_t>(iy) * nx_ + ix]) last = iy;
297 if (iy - last <= my) cells_[static_cast<size_t>(iy) * nx_ + ix] = 1;
298 }
299 last = 1000000;
300 for (int iy = ny_ - 1; iy >= 0; --iy) {
301 if (tmp[static_cast<size_t>(iy) * nx_ + ix]) last = iy;
302 if (last - iy <= my) cells_[static_cast<size_t>(iy) * nx_ + ix] = 1;
303 }
304 }
305 }
306
307 private:
308 double x1_, x2_, y1_, y2_;
309 int nx_, ny_;
310 std::vector<char> cells_;
311};

325void MarkGraph(OccupancyGrid& grid, TGraph* g, const std::string& option) {
326 const int n = g->GetN();
327 const double* x = g->GetX();
328 const double* y = g->GetY();
329 const bool has_marker = HasAny(option, "P*");
330 const bool has_bar = HasAny(option, "B");
331 const bool has_line = HasAny(option, "LCF") || (!has_marker && !has_bar);
332 const bool has_error = option.find('X') == std::string::npos;
333
334 const double marker_px =
335 (g->GetMarkerStyle() == 1 || g->GetMarkerSize() <= 0.0) ? 1.0 : 4.0 * g->GetMarkerSize();
336 const double rx = marker_px / PadWidthPixel();
337 const double ry = marker_px / PadHeightPixel();
338
339 double prev_x = 0.0, prev_y = 0.0;
340 bool has_prev = false;
341 for (int i = 0; i < n; ++i) {
342 double xn = 0.0, yn = 0.0;
343 const bool ok = UserToNdc(x[i], y[i], xn, yn);
344 if (ok) {
345 grid.MarkPoint(xn, yn, has_marker ? rx : 0.0, has_marker ? ry : 0.0);
346 if (has_line && has_prev) grid.MarkSegment(prev_x, prev_y, xn, yn);
347 prev_x = xn;
348 prev_y = yn;
349 }
350 has_prev = ok;
351
352 if (has_bar) {
353 const double base = gPad->GetLogy() != 0 ? std::pow(10.0, gPad->GetUymin()) : 0.0;
354 grid.MarkUserSegment(x[i], base, x[i], y[i]);
355 }
356 if (has_error) {
357 const double exl = g->GetErrorXlow(i), exh = g->GetErrorXhigh(i);
358 const double eyl = g->GetErrorYlow(i), eyh = g->GetErrorYhigh(i);
359 if (eyl > 0.0 || eyh > 0.0) grid.MarkUserSegment(x[i], y[i] - eyl, x[i], y[i] + eyh);
360 if (exl > 0.0 || exh > 0.0) grid.MarkUserSegment(x[i] - exl, y[i], x[i] + exh, y[i]);
361 }
362 }

365void MarkHist1(OccupancyGrid& grid, TH1* h, const std::string& option) {
366 // Skip empty frame histograms such as the one from TPad::DrawFrame.
367 if (h->GetEntries() == 0 && h->GetSumOfWeights() == 0.0) return;
368
369 const bool has_hist = option.find("HIST") != std::string::npos;
370 const bool has_error = !has_hist && option.find('E') != std::string::npos;
371 const bool has_marker = !has_hist && option.find('P') != std::string::npos;
372 const bool has_curve = !has_hist && HasAny(option, "LC");
373 const bool has_step = has_hist || (!has_error && !has_marker && !has_curve);
374 const bool has_fill =
375 option.find('B') != std::string::npos ||
376 (h->GetFillStyle() != 0 && h->GetFillColor() != 0 && h->GetFillStyle() != 4000);
377 const double base = gPad->GetLogy() != 0 ? std::pow(10.0, gPad->GetUymin()) : 0.0;
378
379 const int n = h->GetNbinsX();
380 for (int i = 1; i <= n; ++i) {
381 const double xl = h->GetBinLowEdge(i);
382 const double xh = h->GetBinLowEdge(i + 1);
383 const double xc = h->GetBinCenter(i);
384 const double y = h->GetBinContent(i);
385 if (has_step) {
386 grid.MarkUserSegment(xl, y, xh, y);
387 if (i > 1) grid.MarkUserSegment(xl, h->GetBinContent(i - 1), xl, y);
388 }
389 if (has_fill) grid.MarkUserRect(xl, base, xh, y);
390 if (has_error) {
391 grid.MarkUserSegment(xc, y - h->GetBinErrorLow(i), xc, y + h->GetBinErrorUp(i));
392 }
393 if (has_marker) grid.MarkUserSegment(xc, y, xc, y);
394 if (has_curve && i > 1)
395 grid.MarkUserSegment(h->GetBinCenter(i - 1), h->GetBinContent(i - 1), xc, y);
396 }
397}

409void MarkFunction(OccupancyGrid& grid, TF1* f) {
410 double x_min = gPad->GetUxmin(), x_max = gPad->GetUxmax();
411 if (gPad->GetLogx() != 0) {
412 x_min = std::pow(10.0, x_min);

416 x_max = std::min(x_max, f->GetXmax());

430void MarkText(OccupancyGrid& grid, TText* t) {
431 UInt_t w = 0, h = 0;
432 if (t->InheritsFrom(TLatex::Class())) {
433 static_cast<TLatex*>(t)->GetBoundingBox(w, h);
434 } else {
435 t->GetTextExtent(w, h, t->GetTitle());
436 }
437 double x = t->GetX(), y = t->GetY();
438 if (!t->TestBit(TText::kTextNDC) && !UserToNdc(t->GetX(), t->GetY(), x, y)) return;
439
440 const int h_align = t->GetTextAlign() / 10;
441 const int v_align = t->GetTextAlign() % 10;
442 const double fx = h_align == 2 ? 0.5 : (h_align == 3 ? 1.0 : 0.0);
443 const double fy = v_align == 2 ? 0.5 : (v_align == 3 ? 1.0 : 0.0);
444 const double angle = t->GetTextAngle() * TMath::DegToRad();
445 const double c = std::cos(angle), s = std::sin(angle);
446
447 // Corners in pixels relative to the anchor, rotated, then the bounding box in NDC.
448 double xa = 1e300, xb = -1e300, ya = 1e300, yb = -1e300;
449 for (double u : {-fx * w, (1.0 - fx) * w}) {
450 for (double v : {-fy * h, (1.0 - fy) * h}) {
451 const double px = u * c - v * s;
452 const double py = u * s + v * c;
453 xa = std::min(xa, x + px / PadWidthPixel());
454 xb = std::max(xb, x + px / PadWidthPixel());
455 ya = std::min(ya, y + py / PadHeightPixel());
456 yb = std::max(yb, y + py / PadHeightPixel());
457 }
458 }
459 grid.MarkRect(xa, ya, xb, yb);
460}

462void MarkObject(OccupancyGrid& grid, TObject* obj, const std::string& option,
463 const TObject* ignore) {
464 if (obj == nullptr || obj == ignore) return;
465
466 if (obj->InheritsFrom(TFrame::Class()) || obj->InheritsFrom(TVirtualPad::Class())) {
467 return;
468 } else if (obj->InheritsFrom(TPave::Class())) {
469 TPave* pave = static_cast<TPave*>(obj);
470 grid.MarkRect(pave->GetX1NDC(), pave->GetY1NDC(), pave->GetX2NDC(), pave->GetY2NDC());
471 } else if (obj->InheritsFrom(TBox::Class())) {
472 TBox* box = static_cast<TBox*>(obj);
473 grid.MarkUserRect(box->GetX1(), box->GetY1(), box->GetX2(), box->GetY2());
474 } else if (obj->InheritsFrom(TLine::Class())) {
475 TLine* line = static_cast<TLine*>(obj);
476 if (line->TestBit(TLine::kLineNDC)) {
477 grid.MarkSegment(line->GetX1(), line->GetY1(), line->GetX2(), line->GetY2());
478 } else {
479 grid.MarkUserSegment(line->GetX1(), line->GetY1(), line->GetX2(), line->GetY2());
480 }
481 } else if (obj->InheritsFrom(TMarker::Class())) {
482 TMarker* m = static_cast<TMarker*>(obj);
483 double x = 0.0, y = 0.0;
484 if (UserToNdc(m->GetX(), m->GetY(), x, y)) {
485 const double r = 4.0 * m->GetMarkerSize();
486 grid.MarkPoint(x, y, r / PadWidthPixel(), r / PadHeightPixel());
487 }
488 } else if (obj->InheritsFrom(TText::Class())) {
489 MarkText(grid, static_cast<TText*>(obj));
490 } else if (obj->InheritsFrom(TMultiGraph::Class())) {
491 TMultiGraph* mg = static_cast<TMultiGraph*>(obj);
492 TIter next(mg->GetListOfGraphs());
493 while (TObject* g = next()) {
494 const std::string g_option = ToUpper(next.GetOption());
495 MarkObject(grid, g, g_option.empty() ? option : g_option, ignore);
496 }
497 MarkFunctions(grid, mg->GetListOfFunctions(), ignore);
498 } else if (obj->InheritsFrom(TGraph::Class())) {
499 TGraph* g = static_cast<TGraph*>(obj);
500 MarkGraph(grid, g, option);
501 MarkFunctions(grid, g->GetListOfFunctions(), ignore);
502 } else if (obj->InheritsFrom(THStack::Class())) {
503 THStack* hs = static_cast<THStack*>(obj);
504 TObjArray* stack = option.find("NOSTACK") != std::string::npos ? nullptr : hs->GetStack();
505 TIter next(stack != nullptr ? static_cast<TCollection*>(stack)
506 : static_cast<TCollection*>(hs->GetHists()));
507 while (TObject* h = next()) MarkObject(grid, h, option, ignore);
508 } else if (obj->InheritsFrom(TH2::Class())) {
509 MarkHist2(grid, static_cast<TH2*>(obj));
510 } else if (obj->InheritsFrom(TH1::Class())) {
511 MarkHist1(grid, static_cast<TH1*>(obj), option);
512 MarkFunctions(grid, static_cast<TH1*>(obj)->GetListOfFunctions(), ignore);
513 } else if (obj->InheritsFrom(TF1::Class())) {
514 MarkFunction(grid, static_cast<TF1*>(obj));
515 }

541// Consecutive free cells from each row toward the opposite edge, with rows counted from
542// the top (from_top) or the bottom edge of the frame.
543class FreeRuns {
544 public:
545 FreeRuns(const OccupancyGrid& grid, bool from_top)
546 : nx_(grid.nx()), runs_(static_cast<size_t>(grid.nx()) * grid.ny(), 0) {
547 for (int ix = 0; ix < grid.nx(); ++ix) {
548 int run = 0;
549 for (int r = grid.ny() - 1; r >= 0; --r) {
550 const int iy = from_top ? grid.ny() - 1 - r : r;
551 run = grid.At(ix, iy) ? 0 : run + 1;
552 runs_[static_cast<size_t>(r) * nx_ + ix] = run;
553 }
554 }
555 }
556
557 int At(int ix, int r) const { return runs_[static_cast<size_t>(r) * nx_ + ix]; }
558
559 private:
560 int nx_;
561 std::vector<int> runs_;
562};

564// Free space of a box anchored at a corner and moved `offset` cells away from it along the
565// vertical frame edge (down for top corners, up for bottom corners).
566struct CornerSpace {
567 LegendPosition corner;
568 const OccupancyGrid* grid;
569 int inset_x;
570 int inset_y;
571 int offset;
572 // heights[w - 1] is the free height in cells for a width of w cells; non-increasing.
573 std::vector<int> heights;
574
575 CornerSpace(LegendPosition c, const OccupancyGrid& g, const FreeRuns& runs, int ix0, int iy0,
576 int off)
577 : corner(c), grid(&g), inset_x(ix0), inset_y(iy0), offset(off) {
578 int limit = g.ny() - 2 * inset_y - offset;
579 for (int col = inset_x; col < g.nx() - inset_x && limit > 0; ++col) {
580 const int ix = IsRight(corner) ? g.nx() - 1 - col : col;
581 limit = std::min(limit, runs.At(ix, inset_y + offset));
582 if (limit == 0) break;
583 heights.push_back(limit);
584 }
585 }
586
587 LegendBox Box(double width, double height) const {
588 LegendBox box;
589 box.position = corner;
590 box.shift = offset * grid->CellHeight();
591 const double off_x = inset_x * grid->CellWidth();
592 const double off_y = inset_y * grid->CellHeight() + box.shift;
593 box.x1 = IsRight(corner) ? grid->x2() - off_x - width : grid->x1() + off_x;
594 box.y1 = IsTop(corner) ? grid->y2() - off_y - height : grid->y1() + off_y;
595 box.x2 = box.x1 + width;
596 box.y2 = box.y1 + height;
597 return box;
598 }
599
600 // Largest-area box.
601 LegendBox MaxBox() const {
602 size_t best_w = 0;
603 double best_area = 0.0;
604 for (size_t w = 1; w <= heights.size(); ++w) {
605 const double area = static_cast<double>(w) * heights[w - 1];
606 if (area > best_area) {
607 best_area = area;
608 best_w = w;
609 }
610 }
611 if (best_w == 0) return Box(0.0, 0.0);
612 return Box(best_w * grid->CellWidth(), heights[best_w - 1] * grid->CellHeight());
613 }
614
615 double MaxArea() const {
616 const LegendBox box = MaxBox();
617 return box.Width() * PadWidthPixel() * box.Height() * PadHeightPixel();
618 }
619
620 // Largest scale (<= 1) at which a width x height NDC box fits.
621 double FitScale(double width, double height) const {
622 double best = 0.0;
623 for (size_t w = 1; w <= heights.size(); ++w) {
624 const double s =
625 std::min(w * grid->CellWidth() / width, heights[w - 1] * grid->CellHeight() / height);
626 best = std::max(best, s);
627 if (best >= 1.0) return 1.0;
628 }
629 return best;
630 }
631};

667// Placement of a width x height NDC box. Preference: a corner where it fits at full size
668// (most free area first); then, if allowed, the smallest shift at which it fits; then the
669// largest text scale over all corners and shifts.
670LegendBox FindPlacement(const LegendSpaces& spaces, const std::vector<LegendPosition>& corners,
671 double width, double height, bool allow_shift) {
672 const int max_offset = allow_shift ? spaces.MaxOffset() : 0;
673
674 for (int offset = 0; offset <= max_offset; ++offset) {
675 const CornerSpace* best = nullptr;
676 std::vector<CornerSpace> candidates;
677 candidates.reserve(corners.size());
678 for (LegendPosition corner : corners) candidates.push_back(spaces.At(corner, offset));
679 double best_area = -1.0;
680 for (const CornerSpace& space : candidates) {
681 if (space.FitScale(width, height) < 1.0) continue;
682 const double area = space.MaxArea();
683 if (area > best_area) {
684 best_area = area;
685 best = &space;
686 }
687 }
688 if (best != nullptr) return best->Box(width, height);
689 }
690
691 // Nothing fits at full size: reduce the content as little as possible.
692 LegendBox best_box = spaces.At(corners.front(), 0).Box(width, height);
693 double best_scale = 0.0;
694 for (int offset = 0; offset <= max_offset; ++offset) {
695 for (LegendPosition corner : corners) {
696 const CornerSpace space = spaces.At(corner, offset);
697 const double scale = space.FitScale(width, height);
698 if (scale > best_scale) {
699 best_scale = scale;
700 best_box = space.Box(width * scale, height * scale);
701 best_box.text_scale = scale;
702 }
703 }
704 }
705 return best_box;
706}

708LegendBox MaxBoxOver(const LegendSpaces& spaces, const std::vector<LegendPosition>& corners,
709 bool allow_shift) {
710 LegendBox best;
711 double best_area = 0.0;
712 const int max_offset = allow_shift ? spaces.MaxOffset() : 0;
713 for (int offset = 0; offset <= max_offset; ++offset) {
714 for (LegendPosition corner : corners) {
715 const CornerSpace space = spaces.At(corner, offset);
716 const double area = space.MaxArea();
717 if (area > best_area) {
718 best_area = area;
719 best = space.MaxBox();
720 }
721 }
722 }
723 return best;
724}

767std::pair<double, double> GetLegendContentSize(TLegend* leg, double text_size) {
768 const double w_px = PadWidthPixel();
769 const double h_px = PadHeightPixel();
770 const double text_px = TextHeightPixel(text_size);
771
772 TLatex latex;
773 latex.SetTextFont(leg->GetTextFont());
774 latex.SetTextSize(text_size);
775
776 // Symbol column and gaps follow the text height so the layout scales with the text.
777 const double symbol_px = 2.5 * text_px;
778 const double pad_px = 0.4 * text_px;
779
780 double label_px = 0.0;
781 double header_px = 0.0;
782 int n_entries = 0;
783 int n_headers = 0;
784 TIter next(leg->GetListOfPrimitives());
785 while (TObject* obj = next()) {
786 TLegendEntry* entry = static_cast<TLegendEntry*>(obj);
787 UInt_t w = 0, h = 0;
788 latex.SetTitle(entry->GetLabel());
789 latex.GetBoundingBox(w, h);
790 const std::string opt = ToUpper(entry->GetOption());
791 if (opt.find('H') != std::string::npos) {
792 header_px = std::max(header_px, static_cast<double>(w));
793 ++n_headers;
794 } else {
795 label_px = std::max(label_px, static_cast<double>(w));
796 ++n_entries;
797 }
798 }
799
800 const int n_columns = std::max(1, leg->GetNColumns());
801 const int n_rows = n_headers + (n_entries + n_columns - 1) / n_columns;
802 const double column_px = symbol_px + label_px + pad_px;
803 const double width_px = std::max(n_columns * column_px, header_px + 2.0 * pad_px);
804 const double height_px = std::max(1, n_rows) * 1.3 * text_px;
805
806 // TLegend reserves fMargin of each column for the symbol.
807 leg->SetMargin(static_cast<float>(symbol_px * n_columns / width_px));
808 return {width_px / w_px, height_px / h_px};
809}

879// Tick labels of a linear axis between x_min and x_max as ROOT formats them, with the value
880// at which each one is drawn.
881std::vector<std::pair<double, std::string>> GetLinearAxisLabels(TAxis* axis, double x_min,
882 double x_max) {
883 std::vector<std::pair<double, std::string>> labels;
884 const int n_div_code = axis->GetNdivisions();
885 const int n_primary = std::abs(n_div_code) % 100;
886
887 double bin_low = 0.0;
888 double bin_high = 0.0;
889 double bin_width = 0.0;
890 int n_bins = 0;
891
892 if (n_div_code > 0) {
893 THLimitsFinder::Optimize(x_min, x_max, n_primary, bin_low, bin_high, n_bins, bin_width, "");
894 } else {
895 bin_low = x_min;
896 bin_high = x_max;
897 n_bins = n_primary;
898 bin_width = (x_max - x_min) / static_cast<double>(n_bins);
899 }
900
901 const int max_digits = TGaxis::GetMaxDigits();
902
903 // Find the single global exponent N for the entire axis
904 int global_exp = 0;
905 if (!axis->GetNoExponent()) {
906 const double max_val = std::max(std::abs(bin_low), std::abs(bin_high));
907 if (max_val != 0.0) {
908 if (max_val >= std::pow(10.0, max_digits)) {
909 const int int_digits = static_cast<int>(std::floor(std::log10(max_val))) + 1;
910 const int diff = int_digits - max_digits;
911 if (diff > 0) {
912 global_exp = ((diff + 2) / 3) * 3;
913 }
914 } else if (max_val < std::pow(10.0, -max_digits)) {
915 const int neg_digits = static_cast<int>(std::floor(std::log10(max_val)));
916 int n = neg_digits;
917 if (n % 3 != 0) {
918 n = (n / 3) * 3 - 3;
919 }
920 global_exp = n;
921 }
922 }
923 }
924
925 // Calculate the number of decimals based on the scaled bin width
926 double scaled_bin_width = bin_width;
927 if (global_exp != 0) {
928 scaled_bin_width /= std::pow(10.0, global_exp);
929 }
930
931 int decimals = 0;
932 if (scaled_bin_width < 1.0) {
933 decimals = static_cast<int>(std::ceil(-std::log10(scaled_bin_width) + 1e-9));
934 }
935
936 for (int i = 0; i <= n_bins; ++i) {
937 double val = bin_low + i * bin_width;
938 if (std::abs(val) < 1e-10) {
939 val = 0.0;
940 }
941
942 TString s;
943 double scaled_val = val;
944 if (global_exp != 0) {
945 scaled_val /= std::pow(10.0, global_exp);
946 }
947 s.Form("%.*f", decimals, scaled_val);
948
949 labels.emplace_back(val, s.Data());
950 }
951
952 return labels;
953}

1101// Exponent n of the "x10^{n}" factor that TGaxis draws for a linear axis, or 0 if none.
1102// Follows TGaxis::PaintAxis: the factor is used when the label step is below
1103// 10^-maxdigits, or when the labels need more than maxdigits digits.
1104int GetAxisExponent(TAxis* axis, bool is_y_axis) {
1105 const bool is_log = is_y_axis ? gPad->GetLogy() != 0 : gPad->GetLogx() != 0;
1106 if (is_log || axis->GetNoExponent() || axis->GetLabels() != nullptr) return 0;
1107
1108 const double u_min = is_y_axis ? gPad->GetUymin() : gPad->GetUxmin();
1109 const double u_max = is_y_axis ? gPad->GetUymax() : gPad->GetUxmax();
1110 if (!(u_max > u_min)) return 0;
1111 const double tolerance = 1e-6 * (u_max - u_min);
1112 std::vector<double> values;
1113 for (const auto& label : GetLinearAxisLabels(axis, u_min, u_max)) {
1114 if (label.first >= u_min - tolerance && label.first <= u_max + tolerance) {
1115 values.push_back(label.first);
1116 }
1117 }
1118 if (values.size() < 2) return 0;
1119
1120 const int max_digits = axis->GetMaxDigits() > 0 ? axis->GetMaxDigits() : TGaxis::GetMaxDigits();
1121 const double w_min = values.front();
1122 const double w_max = values.back();
1123 const int n_steps = static_cast<int>(values.size()) - 1;
1124 const double epsilon = 1e-6;
1125 double ww = std::max(std::abs(w_min), std::abs(w_max));
1126 if (ww == 0.0) return 0;
1127
1128 if (std::abs(w_max - w_min) / n_steps < std::pow(10.0, -max_digits)) {
1129 const double af = std::log10(ww) + epsilon;
1130 if (af < 0) {
1131 int iexe = std::abs(static_cast<int>(af));
1132 if (iexe % 3 == 1) iexe += 2;
1133 if (iexe % 3 == 2) iexe += 1;
1134 return -iexe;
1135 }
1136 }
1137
1138 const double af = (ww >= 1 ? std::log10(ww) : std::log10(ww * 0.0001)) + epsilon;
1139 const int nf = static_cast<int>(af) + 1;

1171// Label geometry of TGaxis (ROOT 6.36, font 42), checked against rendered pads.
1172// Linear X labels hang from the frame, their top 0.1 text height below it plus the offset in
1173// pad-height units. Log X labels sit on a baseline 1.15 label sizes (pad-height units) plus
1174// the offset below the frame. Y labels end left of the frame, centered on their ticks:
1175// linear ones 0.07 text height plus the offset (pad-width units) away, log ones 0.08 label
1176// sizes (pad-width units) plus the offset plus about 1.5 pixels away.
1177constexpr double kXLabelTopGapLinear = 0.10;
1178constexpr double kXLabelBaselineLog = 1.15;
1179constexpr double kYLabelRightGapLinear = 0.07;
1180constexpr double kYLabelRightGapLog = 0.08;
1181constexpr double kYLabelRightGapLogPixel = 1.5;
1182// Ink heights in text heights.
1183constexpr double kDigitHeight = 0.67;
1184constexpr double kSuperscriptLabelHeight = 0.85;
1185// Minimum clearance between labels and from the pad edge, in text heights.
1186constexpr double kLabelClearance = 0.15;
1187// Gap between the axis end and the exponent factor, in label sizes.
1188constexpr double kExponentGap = 0.1;
1189// Minimum space between neighbouring tick labels, in text heights.
1190constexpr double kMinLabelSpacing = 0.5;
1191// Fewest primary divisions that crowded labels are reduced to.
1192constexpr int kMinPrimaryDivisions = 2;
1193// Distance per unit title offset, in title sizes along the axis direction.
1194constexpr double kTitleOffsetUnit = 1.6;
1195
1196// Smallest space in pixels between neighbouring tick labels of an axis: along X between the
1197// label boxes, along Y between the digit heights.
1198double MinLabelSpacingPixel(TAxis* axis, bool is_y_axis, double axis_length_px,
1199 double text_px) {
1200 const std::vector<TickLabel> labels = GetDrawnAxisLabels(axis, is_y_axis);
1201 double min_spacing = std::numeric_limits<double>::infinity();
1202 double previous_half = 0.0;
1203 for (std::size_t i = 0; i < labels.size(); ++i) {
1204 const double half = is_y_axis ? 0.5 * kDigitHeight * text_px
1205 : 0.5 * LabelWidthPixel(axis, labels[i].text);
1206 if (i > 0) {
1207 const double distance = (labels[i].position - labels[i - 1].position) * axis_length_px;
1208 min_spacing = std::min(min_spacing, distance - previous_half - half);
1209 }
1210 previous_half = half;
1211 }
1212 return min_spacing;
1213}

1215// Lowers the number of primary divisions of a linear, optimized axis with numeric labels until
1216// neighbouring labels are kMinLabelSpacing text heights apart. Secondary and tertiary
1217// divisions are kept.
1218void ReduceCrowdedDivisions(TAxis* axis, bool is_y_axis, double axis_length_px, double text_px) {
1219 const bool is_log = is_y_axis ? gPad->GetLogy() != 0 : gPad->GetLogx() != 0;
1220 const int n_div = axis->GetNdivisions();
1221 if (is_log || axis->GetLabels() != nullptr || axis->GetLabelSize() <= 0.0 || n_div <= 0) {
1222 return;
1223 }
1224 const int finer = n_div - n_div % 100;
1225 for (int n_primary = n_div % 100; n_primary > kMinPrimaryDivisions; --n_primary) {
1226 if (MinLabelSpacingPixel(axis, is_y_axis, axis_length_px, text_px) >=
1227 kMinLabelSpacing * text_px) {
1228 return;
1229 }
1230 axis->SetNdivisions(finer + n_primary - 1);
1231 }
1232}

1236void AdjustAxisLabelSpace(TAxis* x_axis, TAxis* y_axis) {
1237 if (x_axis == nullptr || y_axis == nullptr || gPad == nullptr) return;
1238 // Axis ranges are needed to locate the labels.
1239 gPad->Update();
1240
1241 // Thin out crowded labels first; the checks below depend on which labels are drawn.
1242 ReduceCrowdedDivisions(
1243 x_axis, false, (1.0 - gPad->GetLeftMargin() - gPad->GetRightMargin()) * PadWidthPixel(),
1244 TextHeightPixel(x_axis->GetLabelSize()));
1245 ReduceCrowdedDivisions(
1246 y_axis, true, (1.0 - gPad->GetBottomMargin() - gPad->GetTopMargin()) * PadHeightPixel(),
1247 TextHeightPixel(y_axis->GetLabelSize()));
1248
1249 const double w_px = PadWidthPixel();
1250 const double h_px = PadHeightPixel();
1251 const double tx = TextHeightPixel(x_axis->GetLabelSize());
1252 const double ty = TextHeightPixel(y_axis->GetLabelSize());
1253
1254 // The first X label and the lowest Y label can collide at the bottom-left corner.
1255 const std::vector<TickLabel> x_labels = GetDrawnAxisLabels(x_axis, false);
1256 const std::vector<TickLabel> y_labels = GetDrawnAxisLabels(y_axis, true);
1257 if (!x_labels.empty() && !y_labels.empty() && x_axis->GetLabelSize() > 0.0 &&
1258 y_axis->GetLabelSize() > 0.0) {
1259 const TickLabel& x_first = x_labels.front();
1260 const TickLabel& y_first = y_labels.front();
1261 const double frame_left = gPad->GetLeftMargin() * w_px;
1262 const double frame_bottom = gPad->GetBottomMargin() * h_px;
1263 const double frame_w = (1.0 - gPad->GetLeftMargin() - gPad->GetRightMargin()) * w_px;
1264 const double frame_h = (1.0 - gPad->GetBottomMargin() - gPad->GetTopMargin()) * h_px;
1265
1266 const double x_left =
1267 frame_left + x_first.position * frame_w - 0.5 * LabelWidthPixel(x_axis, x_first.text);
1268 const double x_height = x_first.has_superscript ? kSuperscriptLabelHeight : kDigitHeight;
1269 const double y_height = y_first.has_superscript ? kSuperscriptLabelHeight : kDigitHeight;
1270 const double x_top =
1271 gPad->GetLogx() == 0
1272 ? frame_bottom - kXLabelTopGapLinear * tx - x_axis->GetLabelOffset() * h_px
1273 : frame_bottom -
1274 (kXLabelBaselineLog * x_axis->GetLabelSize() + x_axis->GetLabelOffset()) * h_px +
1275 x_height * tx;
1276 const double y_right =
1277 gPad->GetLogy() == 0
1278 ? frame_left - kYLabelRightGapLinear * ty - y_axis->GetLabelOffset() * w_px
1279 : frame_left -
1280 (kYLabelRightGapLog * y_axis->GetLabelSize() + y_axis->GetLabelOffset()) * w_px -
1281 kYLabelRightGapLogPixel;
1282 const double y_bottom = frame_bottom + y_first.position * frame_h - 0.5 * y_height * ty;
1283
1284 const double clearance = kLabelClearance * std::max(tx, ty);
1285 const double overlap_v = x_top - y_bottom + clearance;
1286 const double overlap_h = y_right - x_left + clearance;
1287 if (overlap_v > 0.0 && overlap_h > 0.0) {
1288 // Moving X labels down and Y labels left by the same distance separates them once the
1289 // distance reaches the smaller overlap.
1290 const double shift = std::min(overlap_v, overlap_h);
1291 x_axis->SetLabelOffset(x_axis->GetLabelOffset() + shift / h_px);
1292 x_axis->SetTitleOffset(x_axis->GetTitleOffset() +
1293 shift / (kTitleOffsetUnit * x_axis->GetTitleSize() * h_px));
1294 gPad->SetBottomMargin(gPad->GetBottomMargin() + shift / h_px);
1295 y_axis->SetLabelOffset(y_axis->GetLabelOffset() + shift / w_px);
1296 OptimizeYAxisLayout(y_axis, shift / w_px);
1297 }
1298 }
1299
1300 // Widen the right margin when the last X label would extend past the pad edge.
1301 if (!x_labels.empty() && x_axis->GetLabelSize() > 0.0) {
1302 const TickLabel& x_last = x_labels.back();
1303 const double left = gPad->GetLeftMargin();
1304 const double half_w = 0.5 * LabelWidthPixel(x_axis, x_last.text) / w_px;
1305 const double limit = 1.0 - kLabelClearance * tx / w_px - half_w;
1306 if (x_last.position > 0.0) {
1307 const double frame_right = left + (limit - left) / x_last.position;
1308 const double required_margin = 1.0 - frame_right;
1309 if (required_margin > gPad->GetRightMargin()) gPad->SetRightMargin(required_margin);
1310 }
1311 }
1312
1313 // TGaxis draws the X exponent right of the axis end (0.1 label size away, bottom-left
1314 // aligned on the axis) and the Y exponent above the axis end, plus the style offsets.
1315 if (x_axis->GetLabelSize() > 0.0) {
1316 const int exponent = GetAxisExponent(x_axis, false);
1317 if (exponent != 0) {
1318 double w = 0.0, h = 0.0;
1319 ExponentSizePixel(x_axis, exponent, w, h);
1320 Float_t x_off = 0.0, y_off = 0.0;
1321 gStyle->GetExponentOffset(x_off, y_off, "x");
1322 const double required_margin =
1323 kExponentGap * x_axis->GetLabelSize() + x_off + (w + kLabelClearance * tx) / w_px;
1324 if (required_margin > gPad->GetRightMargin()) gPad->SetRightMargin(required_margin);
1325 }
1326 }
1327 if (y_axis->GetLabelSize() > 0.0) {
1328 const int exponent = GetAxisExponent(y_axis, true);
1329 if (exponent != 0) {
1330 double w = 0.0, h = 0.0;
1331 ExponentSizePixel(y_axis, exponent, w, h);
1332 Float_t x_off = 0.0, y_off = 0.0;
1333 gStyle->GetExponentOffset(x_off, y_off, "y");
1334 const double required_margin =
1335 kExponentGap * y_axis->GetLabelSize() + y_off + (h + kLabelClearance * ty) / h_px;
1336 if (required_margin > gPad->GetTopMargin()) gPad->SetTopMargin(required_margin);
1337 }
1338 }
