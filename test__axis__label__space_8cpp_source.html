1// Checks that the last X label is kept inside the pad and that the first X label and the
2// lowest Y label are separated at the bottom-left corner.
3
4#include <TCanvas.h>
5#include <TGraph.h>
6#include <TROOT.h>
7#include <TStyle.h>
8#include <roothelper/graphics.h>
9
10#include <cmath>
11#include <cstdio>
12
13int main() {
14 gROOT->SetBatch(true);
15 roothelper::Prepare();
16 bool passed = true;
17
18 // Range 0..10 puts labels on both ends of the X axis and "0" at the corner of both axes.
19 TCanvas* c = roothelper::CreateCanvas("c_label_space", "c_label_space");
20 TGraph* g = new TGraph();
21 g->SetPoint(0, 0.0, 0.0);
22 g->SetPoint(1, 10.0, 10.0);
23 g->Draw("AL");
24 g->GetXaxis()->SetLimits(0.0, 10.0);
25 g->SetMinimum(0.0);
26 g->SetMaximum(10.0);
27 roothelper::SetAxes(g);
28
29 if (!(gPad->GetRightMargin() > roothelper::GraphicsSize::current.right_margin)) {
30 std::printf("FAIL: right margin was not widened (%g)\n", gPad->GetRightMargin());
31 passed = false;
32 }
33
34 const double w_px = gPad->GetWw() * gPad->GetAbsWNDC();
35 const double h_px = gPad->GetWh() * gPad->GetAbsHNDC();
36 const double x_shift_px = (g->GetXaxis()->GetLabelOffset() - gStyle->GetLabelOffset("X")) * h_px;
37 const double y_shift_px = (g->GetYaxis()->GetLabelOffset() - gStyle->GetLabelOffset("Y")) * w_px;
38 if (!(x_shift_px > 0.0) || std::abs(x_shift_px - y_shift_px) > 1e-6) {
39 std::printf("FAIL: label shifts differ or are zero (x %g px, y %g px)\n", x_shift_px,
40 y_shift_px);
41 passed = false;
42 }
43 if (!(gPad->GetBottomMargin() > roothelper::GraphicsSize::current.bottom_margin)) {
44 std::printf("FAIL: bottom margin did not follow the X label shift\n");
45 passed = false;
46 }
47 c->Print("axis_label_space.png");
48
49 // Applying the styling again must give the same layout.
50 const double right = gPad->GetRightMargin();
51 const double x_offset = g->GetXaxis()->GetLabelOffset();
52 roothelper::SetAxes(g);
53 if (std::abs(gPad->GetRightMargin() - right) > 1e-9 ||
54 std::abs(g->GetXaxis()->GetLabelOffset() - x_offset) > 1e-9) {
55 std::printf("FAIL: styling twice changed the layout\n");
56 passed = false;
57 }
58
59 // Short labels keep the divisions; five-character labels every 0.005 over 0.05 are reduced.
60 if (g->GetXaxis()->GetNdivisions() % 100 != 10) {
61 std::printf("FAIL: divisions of uncrowded labels changed (%d)\n",
62 g->GetXaxis()->GetNdivisions());
63 passed = false;
64 }
65 TCanvas* cd = roothelper::CreateCanvas("c_label_crowded", "c_label_crowded");
66 TGraph* gd = new TGraph();
67 gd->SetPoint(0, 0.709, 0.0);
68 gd->SetPoint(1, 0.759, 5.0);
69 gd->Draw("AL");
70 gd->GetXaxis()->SetLimits(0.709, 0.759);
71 roothelper::SetAxes(gd);
72 const int crowded_div = gd->GetXaxis()->GetNdivisions();
73 if (!(crowded_div % 100 < 10) || crowded_div / 100 != 5) {
74 std::printf("FAIL: crowded X divisions %d (want fewer primary, secondary kept)\n",
75 crowded_div);
76 passed = false;
77 }
78 cd->Print("axis_label_crowded.png");

80 // Exponent factors: X needs room right of the frame, Y above it.
81 struct ExponentCase {
82 const char* name;
83 double x_max;
84 double y_max;
85 bool x_exponent;
86 bool y_exponent;
87 };
88 const ExponentCase cases[] = {{"none", 10.0, 10.0, false, false},
89 {"x_large", 2e6, 10.0, true, false},
90 {"x_small", 3e-6, 10.0, true, false},
91 {"y_large", 10.0, 2e6, false, true},
92 {"y_small", 10.0, 3e-6, false, true}};
93 for (const ExponentCase& e : cases) {
94 TCanvas* ce = roothelper::CreateCanvas(e.name, e.name);
95 TGraph* ge = new TGraph();
96 ge->SetPoint(0, 0.0, 0.0);
97 ge->SetPoint(1, e.x_max, e.y_max);
98 ge->Draw("AL");
99 ge->GetXaxis()->SetLimits(0.0, e.x_max);
100 ge->SetMinimum(0.0);
101 ge->SetMaximum(e.y_max);
102 roothelper::SetAxes(ge);
103 // "x10^{n}" needs more than the last-label clipping alone (about 0.024 here).
104 const bool wide_right = gPad->GetRightMargin() > 0.06;
105 const bool tall_top = gPad->GetTopMargin() > roothelper::GraphicsSize::current.top_margin;
106 if (wide_right != e.x_exponent || tall_top != e.y_exponent) {
107 std::printf("FAIL: %s: right margin %g, top margin %g\n", e.name, gPad->GetRightMargin(),
108 gPad->GetTopMargin());
109 passed = false;
110 }
111 ce->Print(Form("axis_exponent_%s.png", e.name));
112 }
113
114 if (!passed) return 1;
115 std::printf("Axis label space checks passed.\n");
116 return 0;
117}
