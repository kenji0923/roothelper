41// True if the legend overlaps any graph point or the segments between points.
42bool OverlapsGraph(TLegend* leg, TGraph* g) {
43 for (int i = 0; i + 1 < g->GetN(); ++i) {
44 for (int k = 0; k <= 50; ++k) {
45 const double t = k / 50.0;
46 const double x = g->GetX()[i] + t * (g->GetX()[i + 1] - g->GetX()[i]);
47 const double y = g->GetY()[i] + t * (g->GetY()[i + 1] - g->GetY()[i]);
48 const double xn = (gPad->XtoPad(x) - gPad->GetX1()) / (gPad->GetX2() - gPad->GetX1());
49 const double yn = (gPad->YtoPad(y) - gPad->GetY1()) / (gPad->GetY2() - gPad->GetY1());
50 if (xn > leg->GetX1NDC() && xn < leg->GetX2NDC() && yn > leg->GetY1NDC() &&
51 yn < leg->GetY2NDC()) {
52 return true;
53 }
54 }
55 }
56 return false;
57}

71void TestRisingCurve() {
72 // A line from bottom-left to top-right leaves the other two corners free.
73 TCanvas* c = roothelper::CreateCanvas("c_rising", "c_rising");
74 TGraph* g = MakeGraph("Rising", [](double x) { return x; });
75 g->Draw("AL");
76 roothelper::SetAxes(g);
77 TLegend* leg = roothelper::PutLegend();
78 const roothelper::LegendPosition p =
79 roothelper::FindMaxLegendBox(roothelper::kLegendMargin, leg).position;
80 Check(p == roothelper::LegendPosition::TopLeft || p == roothelper::LegendPosition::BottomRight,
81 std::string("rising: best corner is ") + CornerName(p));
82 Check(!OverlapsGraph(leg, g), "rising: legend overlaps graph");
83 Check(std::abs(leg->GetTextSize() - roothelper::GraphicsSize::current.text_size) < 1e-9,
84 "rising: text size changed although content fits");
85 Check(leg->GetX2NDC() - leg->GetX1NDC() < 0.5, "rising: legend was not shrunk to content");
86 c->Print("legend_rising.png");
87}

89void TestSameObjects() {
90 // The first graph leaves the top free; a graph drawn with SAME fills the top-left.
91 TCanvas* c = roothelper::CreateCanvas("c_same", "c_same");
92 TH1D* h = new TH1D("h_same", "Histogram;x;y", 50, 0.0, 10.0);
93 for (int i = 1; i <= 50; ++i) h->SetBinContent(i, 1.0 + 0.2 * std::sin(i * 0.3));
94 h->SetMaximum(10.0);
95 h->SetMinimum(0.0);
96 h->Draw("HIST");
97 roothelper::SetAxes(h);
98 TGraph* g = MakeGraph("Falling", [](double x) { return 9.5 - 0.9 * x; });
99 g->Draw("L SAME");
100 TLegend* leg = roothelper::PutLegend();
101 const bool at_top_right = std::abs(leg->GetX2NDC() - (1.0 - gPad->GetRightMargin())) < 0.05 &&
102 std::abs(leg->GetY2NDC() - (1.0 - gPad->GetTopMargin())) < 0.05;
103 Check(at_top_right, "same: legend is not at top right");
104 Check(!OverlapsGraph(leg, g), "same: legend overlaps the SAME graph");
105 c->Print("legend_same.png");
106}

108void TestShrinkText() {
109 // A long legend in a crowded pad needs a smaller text.

137void TestShiftFromCorner() {
138 // Lines near the top and the bottom leave only the middle free: no corner has room, so the
139 // legend moves down from a top corner (or up from a bottom one) instead of shrinking.
140 TCanvas* c = roothelper::CreateCanvas("c_shift", "c_shift");
141 TGraph* upper = MakeGraph("Upper line", [](double x) { return 9.0 + 0.05 * std::sin(x); });
142 TGraph* lower = MakeGraph("Lower line", [](double x) { return 1.0 + 0.05 * std::sin(x); });
143 upper->Draw("AL");
144 upper->SetMinimum(0.0);
145 upper->SetMaximum(10.0);
146 roothelper::SetAxes(upper);
147 lower->Draw("L SAME");
148
149 roothelper::LegendBox corner_box =
150 roothelper::FindMaxLegendBox(roothelper::kLegendMargin, nullptr, false);
151 roothelper::LegendBox placement;
152 TLegend* leg = roothelper::PutLegend("", roothelper::kLegendMargin, &placement);
153 std::printf(
154 "shift: corner max box height %.3f; legend at %s, shift %.3f, scale %.3f,"
155 " (%.3f, %.3f)-(%.3f, %.3f)\n",
156 corner_box.Height(), CornerName(placement.position), placement.shift, placement.text_scale,
157 placement.x1, placement.y1, placement.x2, placement.y2);
158 Check(placement.IsShifted(), "shift: legend was not moved from the corner");
159 Check(placement.text_scale == 1.0, "shift: text was reduced although moving fits");
160 Check(std::abs(leg->GetY2NDC() - placement.y2) < 1e-12 &&
161 std::abs(leg->GetX1NDC() - placement.x1) < 1e-12,
162 "shift: reported coordinates differ from the legend");
163 Check(!OverlapsGraph(leg, upper) && !OverlapsGraph(leg, lower), "shift: legend overlaps a line");
164
165 // With moving disabled the same legend must shrink.
166 roothelper::LegendBox no_shift;
167 roothelper::PlaceLegend(leg, roothelper::kLegendMargin,
168 roothelper::GraphicsSize::current.text_size, &no_shift, false);
169 Check(!no_shift.IsShifted() && no_shift.text_scale < 1.0,
170 "shift: without moving the text was not reduced");
171 roothelper::PlaceLegend(leg); // Restore the moved placement for the picture.
172 c->Print("legend_shift.png");
173}
