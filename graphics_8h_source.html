351template <class GraphType>
352void SetZAxis(GraphType* graph_object) {
353 TObject* obj = dynamic_cast<TObject*>(graph_object);
354 if (obj == nullptr) return;
355
356 TAxis* axis_for_attributes = nullptr;
357 TAxis* axis_for_limits = nullptr;
358
359 if (obj->InheritsFrom(TH1::Class())) {
360 axis_for_attributes = dynamic_cast<TH1*>(obj)->GetZaxis();
361 axis_for_limits = axis_for_attributes;
362 } else if (obj->InheritsFrom(TGraph2D::Class())) {
363 TGraph2D* g2d = dynamic_cast<TGraph2D*>(obj);
364 axis_for_attributes = g2d->GetZaxis();
365 if (g2d->GetHistogram() != nullptr) {
366 axis_for_limits = g2d->GetHistogram()->GetZaxis();
367 } else {
368 axis_for_limits = g2d->GetZaxis();
369 }
370 }
371
372 if (axis_for_attributes == nullptr || axis_for_limits == nullptr) return;
373
374 double z_min = 0.0;
375 double z_max = 0.0;
376 if (obj->InheritsFrom(TH1::Class())) {
377 TH1* h = dynamic_cast<TH1*>(obj);
378 z_min = h->GetMinimum();
379 z_max = h->GetMaximum();
380 } else if (obj->InheritsFrom(TGraph2D::Class())) {
381 TGraph2D* g2d = dynamic_cast<TGraph2D*>(obj);
382 if (g2d->GetHistogram() != nullptr) {
383 z_min = g2d->GetHistogram()->GetMinimum();
384 z_max = g2d->GetHistogram()->GetMaximum();
385 }
386 }
387
388 axis_for_attributes->SetTitleSize(GraphicsSize::current.text_size);
389 axis_for_attributes->SetLabelSize(GraphicsSize::current.text_size);
390 axis_for_attributes->SetTitleOffset(GraphicsSize::current.title_offset_x);
391 axis_for_attributes->SetDecimals(true);
392 axis_for_attributes->CenterTitle();
393
394 if (axis_for_limits != axis_for_attributes) {
395 axis_for_limits->SetTitleSize(GraphicsSize::current.text_size);
396 axis_for_limits->SetLabelSize(GraphicsSize::current.text_size);
397 axis_for_limits->SetTitleOffset(GraphicsSize::current.title_offset_x);
398 axis_for_limits->SetDecimals(true);
399 axis_for_limits->CenterTitle();
400 }
401
402 TPaletteAxis* palette = nullptr;
403 if (obj->InheritsFrom(TH1::Class())) {
404 TObject* pal_obj = dynamic_cast<TH1*>(obj)->GetListOfFunctions()->FindObject("palette");
405 if (pal_obj != nullptr) {
406 palette = static_cast<TPaletteAxis*>(pal_obj);
407 }
408 } else if (obj->InheritsFrom(TGraph2D::Class())) {
409 TGraph2D* g2d = dynamic_cast<TGraph2D*>(obj);
410 if (g2d->GetHistogram() != nullptr) {
411 TObject* pal_obj = g2d->GetHistogram()->GetListOfFunctions()->FindObject("palette");
412 if (pal_obj != nullptr) {
413 palette = static_cast<TPaletteAxis*>(pal_obj);
414 }
415 }
416 }
417
418 if (palette != nullptr) {
419 TGaxis* p_axis = palette->GetAxis();
420 if (p_axis != nullptr) {
421 p_axis->SetTitleSize(GraphicsSize::current.text_size);
422 p_axis->SetLabelSize(GraphicsSize::current.text_size);
423
424 const double label_width = GetMaxLabelWidthNdc(axis_for_limits, true, false, z_min, z_max);
425 const double tick_length = p_axis->GetTickSize();
426 const double title_size = p_axis->GetTitleSize();
427 const double gap = 0.010;
428 const double distance_from_axis = tick_length + label_width + gap;
429 double title_offset = (distance_from_axis / title_size) * 0.60 + 0.20;
430 if (title_offset < 0.3) {
431 title_offset = 0.3;
432 }
433
434 axis_for_attributes->SetTitleOffset(title_offset);
435 axis_for_limits->SetTitleOffset(title_offset);
436 p_axis->SetTitleOffset(title_offset);
437 palette->SetTitleOffset(title_offset);
438 axis_for_attributes->CenterTitle();
439 axis_for_limits->CenterTitle();
440 p_axis->CenterTitle();
441
442 // Check if Z-axis requires an exponent multiplier at the top of the axis
443 bool has_exponent = false;
444 if (gPad->GetLogz() == 0 && !axis_for_limits->GetNoExponent()) {
445 const double max_val = std::max(std::abs(z_min), std::abs(z_max));
446 const int max_digits = TGaxis::GetMaxDigits();
447 if (max_val != 0.0 &&
448 (max_val >= std::pow(10.0, max_digits) || max_val < std::pow(10.0, -max_digits))) {
449 has_exponent = true;
450 }
451 }
452
453 const double right_edge_buffer = 0.010;
454 double required_margin = 0.045 + distance_from_axis + title_size + right_edge_buffer;
455
456 if (gPad->GetRightMargin() < required_margin) {
457 gPad->SetRightMargin(required_margin);
458 }
459
460 if (has_exponent) {
461 const double required_top_margin = 0.065;
462 if (gPad->GetTopMargin() < required_top_margin) {
463 gPad->SetTopMargin(required_top_margin);
464 }
465 }
466 }
467
468 palette->SetX1NDC(1.0 - gPad->GetRightMargin() + 0.01);
469 palette->SetX2NDC(1.0 - gPad->GetRightMargin() + 0.045);
470 palette->SetY1NDC(gPad->GetBottomMargin());
471 palette->SetY2NDC(1.0 - gPad->GetTopMargin());
472 }

557template <class ObjectType>
558std::vector<TCanvas*> DrawWithAutoRecreatorOfCanvas(const char* canvas_name_title, size_t n_pad_x,
559 size_t n_pad_y,
560 const std::vector<ObjectType*>& object_list,
561 const char* draw_option) {
562 std::vector<TCanvas*> c_list;
563
564 const size_t n_pad = n_pad_x * n_pad_y;
565 size_t current_pad = n_pad + 1;
566 for (size_t i = 0; i < object_list.size(); ++i) {
567 TCanvas* c = nullptr;
568 if (current_pad > n_pad) {
569 c_list.push_back(new TCanvas(Form("%s_%zu", canvas_name_title, c_list.size()),
570 Form("%s_%zu", canvas_name_title, c_list.size()), 700 * n_pad_x,
571 500 * n_pad_y));
572 c = c_list.back();
573 c->Divide(n_pad_x, n_pad_y);
574 current_pad = 1;
575 } else {
576 c = c_list.back();
577 }
578 c->cd(current_pad);
579 object_list[i]->Draw(draw_option);
580 ++current_pad;
581 }
582
583 return c_list;
584}

615// TLatex draws Greek codes such as "#mu" from the (serif, slanted) Symbol font.
616// These Latin-1 bytes are drawn in the *current* text font instead, so with font
617// 42 they render upright and sans-serif. Prefer these namespaced constants to a

620// string at run time, e.g.:

622inline constexpr char kMicro[] = "\xB5"; // micro sign U+00B5
